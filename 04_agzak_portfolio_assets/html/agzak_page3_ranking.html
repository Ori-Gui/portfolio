@Scheduled(cron = "55 59 * * * *", zone = "Asia/Seoul")
@SchedulerLock(name = "rankingSnapshot",
               lockAtMostFor = "PT4M", lockAtLeastFor = "PT20S")
public void todayRebuild() { rankingService.rebuildTodaySnapshot(); }

@Transactional
public void rebuildTodaySnapshot() {
    LocalDate today = redisStore.today();
    var all = redisStore.readTop(today, redisStore.zTotal(today),
                                  Integer.MAX_VALUE);

    // ★ 탈퇴 회원 prune — Redis 고아 데이터 정리
    Set<Long> ids   = all.stream().map(UserRow::memberId).collect(toSet());
    Set<Long> alive = mysqlStore.existingMemberIds(ids);
    for (Long mid : ids)
        if (!alive.contains(mid)) redisStore.removeTodayUser(mid);

    // 살아있는 회원만 rank_no 1번부터 영속 저장
    int rank = 1;
    List<Ranking> rows = new ArrayList<>(alive.size());
    for (var r : all) {
        if (!alive.contains(r.memberId())) continue;
        rows.add(Ranking.builder()
            .member(mysqlStore.refMember(r.memberId()))   // em.getReference (N+1 회피)
            .date(today).rankNo(rank++)
            .totalScore(r.total()).originalTotalScore(r.original())
            .build());
    }
    mysqlStore.replaceTodaySnapshot(today, rows);   // deleteByDate + saveAll
}
