public Page<ElasticSearchResponseDto> searchByKeyword(String keyword, Long categoryId, Pageable pageable, String sort) {
    var req = new SearchRequest.Builder()
        .index(elasticConfig.getBookIndexName())   // "ejoping-books"
        .from((int) pageable.getOffset())
        .size(pageable.getPageSize())
        .query(q -> q.bool(b -> {

            // ① 본문 필드 multi_match — 가중치 + 4종 분석기 활용
            b.should(s -> s.multiMatch(m -> m.query(keyword).fields(
                "book_title^10",    "book_title.ngram^3",
                "book_title.jaso",   "book_title.synonym",
                "book_desc^3",       "book_desc.ngram",    "book_desc.synonym",
                "book_publisher^2",  "book_publisher.ngram", "book_publisher.synonym"
            )));

            // ② 카테고리 nested — 트리 구조 책 분류 검색
            b.should(s -> s.nested(n -> n.path("book_category").query(
                cq -> cq.multiMatch(m -> m.query(keyword).fields(
                    "book_category.name^3", "book_category.name.ngram", "book_category.name.synonym")))));

            // ③ 태그 nested · ④ 기여자 nested (저자/번역/출판) 동일 패턴
            // b.should(...book_tag...);  b.should(...book_contributor...);

            return b;
        }));

    // 정렬 — popularity / newest / lowest_price / highest_price / rating / reviews
    sortCriteria.forEach(e -> req.sort(s -> s.field(f -> f.field(e.getKey()).order(...))));

    // → 평균 280ms · Slow Query 0건
    return elasticsearchClient.search(req.build(), Map.class);
}
