UserDetails + Principal 동시 구현 
@Getter @RequiredArgsConstructor
public class JwtUserDetails
        implements UserDetails, Principal {

    private final Long userId;
    private final String nickname;
    private final Collection<? extends GrantedAuthority> authorities;

    // ★ Principal.getName() == userId — 1:1 라우팅 키
    @Override
    public String getName() {
        return String.valueOf(userId);
    }

    // UserDetails 메서드들 (생략)
    @Override public String getUsername() { return getName(); }
    @Override public String getPassword() { return null; }
}

// JwtHandshakeInterceptor — HTTP 업그레이드 단계
public boolean beforeHandshake(ServerHttpRequest req, ...,
                                Map<String, Object> attributes) {
    String token = extractToken(req);
    JwtUserDetails principal = jwtProvider.parse(token);
    attributes.put("user", principal);   // ★ STOMP 단계로 전달
    return true;
}

// WsAuthChannelInterceptor — STOMP CONNECT 단계
@Override
public Message<?> preSend(Message<?> msg, MessageChannel ch) {
    StompHeaderAccessor acc = StompHeaderAccessor.wrap(msg);
    if (StompCommand.CONNECT.equals(acc.getCommand())) {
        JwtUserDetails p = (JwtUserDetails) acc.getSessionAttributes().get("user");
        acc.setUser(p);   // ★ Principal.getName() = userId 일관 주입
    }
    return msg;
}
