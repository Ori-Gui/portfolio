@RestController @RequiredArgsConstructor
@RequestMapping("/auth/kakao")
public class AuthController {

    private final KakaoOAuthService kakaoService;
    private final JwtProvider jwtProvider;

    // ① 앱이 호출 — 카카오 인증 URL 생성
    @GetMapping("/authorize-url")
    public AuthorizeUrlResponse authorizeUrl() {
        return new AuthorizeUrlResponse(kakaoService.buildAuthorizeUrl());
    }

    // ③~⑦ 카카오 콜백 — Code 교환 + 딥링크 복귀
    @GetMapping("/callback")
    public ResponseEntity<Void> callback(@RequestParam String code) {

        // 1. Authorization Code → Access Token 교환
        KakaoTokenResponse kakaoToken = kakaoService.exchangeToken(code);

        // 2. 카카오 프로필 조회 → 회원 조회/생성
        Member member = kakaoService.findOrCreate(kakaoToken);

        // 3. JWT 발급 (Access + Refresh, RTR)
        String accessToken  = jwtProvider.issueAccess(member);
        String refreshToken = jwtProvider.issueRefresh(member);

        // 4. Refresh Token Redis 저장 (TTL)
        jwtProvider.saveRefresh(member.getId(), refreshToken);

        // 5. ★ 커스텀 스킴 딥링크로 복귀 — 앱이 자동 캐치
        String deepLink = "toktokggubi://oauth/kakao"
            + "?accessToken=" + accessToken
            + "&refreshToken=" + refreshToken;

        return ResponseEntity.status(HttpStatus.SEE_OTHER)
            .location(URI.create(deepLink))
            .build();
    }

    // Refresh Token Rotation — 탈취 시나리오 방어
    @PostMapping("/token/refresh")
    public TokenResponse refresh(@RequestBody RefreshRequest req) {
        return jwtProvider.rotate(req.refreshToken());
    }
}
▎ APP RETURN
커스텀 스킴 딥링크
toktokggubi://oauth/kakao 로 앱이 등록한 스킴 사용. 외부 브라우저 → 앱 자동 복귀로 인증 흐름 단절 없음.
▎ JWT FLOW
Access · Refresh 분리 + RTR
Access는 짧은 TTL · Refresh는 Redis TTL 보관. Refresh 쓰는 순간 새 토큰 발급 + 이전 무효화로 탈취 시나리오 방어.
▎ FILTER + REVOCATION
JwtAuthenticationFilter
요청마다 JWT 검증 + auth:blacklist:{jti} 확인. permitAll 경로 정책으로 OAuth2 콜백 인증 우회.
