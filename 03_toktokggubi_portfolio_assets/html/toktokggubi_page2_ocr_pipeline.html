@Component @RequiredArgsConstructor
public class GifticonTextParser {

    private final FranchiseCanonicalMap canonicalMap;

    // 만료일 정규식 — 다양한 형식 통합 파싱
    private static final Pattern EXPIRES_RE = Pattern.compile(
        "(\\d{4})[\\.\\-/년]\\s*(\\d{1,2})[\\.\\-/월]\\s*(\\d{1,2})"
    );

    public GifticonParseResult parse(String ocrText) {
        // 1️⃣ 브랜드 추출 + 캐노니컬 매핑
        String rawBrand = extractBrand(ocrText);
        Long franchiseId = canonicalMap.resolve(rawBrand);
        // "Starbucks" / "스타벅스" / "스벅" → 동일 ID

        // 2️⃣ 만료일 LocalDate 파싱
        LocalDate expiresAt = parseExpiresAt(ocrText);

        // 3️⃣ 상품명 — 브랜드·만료일 제외 후 추출
        String label = extractLabel(ocrText, rawBrand);

        return new GifticonParseResult(franchiseId, label, expiresAt);
    }

    private LocalDate parseExpiresAt(String text) {
        var m = EXPIRES_RE.matcher(text);
        if (!m.find()) return null;
        return LocalDate.of(
            Integer.parseInt(m.group(1)),
            Integer.parseInt(m.group(2)),
            Integer.parseInt(m.group(3))
        );
    }
}

// 캐노니컬 매핑 사전
public class FranchiseCanonicalMap {
    public Long resolve(String raw) {
        return switch (normalize(raw)) {
            case "스타벅스", "starbucks", "스벅" -> STARBUCKS_ID;
            case "이디야", "ediya"           -> EDIYA_ID;
            default -> findBySimilarity(raw);
        };
    }
}
